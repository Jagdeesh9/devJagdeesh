import { Application, Assets, Graphics, Sprite, Text, TextStyle } from "pixi.js"; (async () => { const app = new Application(); await app.init({ background: "#1099bb", resizeTo: window }); document.getElementById("pixi-container")!.appendChild(app.canvas); let isJumping = false; let velocityY = 0; const gravity = 1; // Load Dino Texture const texture = await Assets.load("/assets/bunny.png"); // Replace with actual Dino texture const bunny = new Sprite(texture); bunny.x = 50; bunny.y = 370; bunny.width = 40; bunny.height = 40; app.stage.addChild(bunny); // Ground const ground = new Graphics(); ground.beginFill(0x888888); ground.drawRect(0, 410, app.screen.width, 10); ground.endFill(); app.stage.addChild(ground); // Obstacles array const obstacles: Graphics[] = []; function spawnObstacle() { const type = Math.floor(Math.random() * 3); // 0, 1, or 2 const obs = new Graphics(); switch (type) { case 0: // Short cactus obs.beginFill(0x228822); obs.drawRect(0, 0, 20, 40); obs.endFill(); obs.y = 370; break; case 1: // Tall cactus obs.beginFill(0x116611); obs.drawRect(0, 0, 30, 60); obs.endFill(); obs.y = 350; break; case 2: // Bird obs.beginFill(0x000000); obs.drawPolygon([0, 0, 20, 10, 0, 20]); // triangle shape obs.endFill(); obs.y = 300; break; } obs.x = app.screen.width; app.stage.addChild(obs); obstacles.push(obs); } setInterval(spawnObstacle, 2000); // Jump handling window.addEventListener("keydown", (e) => { if ((e.code === "Space" || e.code === "ArrowUp") && !isJumping) { velocityY = -15; isJumping = true; } }); // Score let score = 0; const style = new TextStyle({ fill: "#000", fontSize: 20 }); const scoreText = new Text(`Score: 0`, style); scoreText.position.set(10, 10); app.stage.addChild(scoreText); // Game loop app.ticker.add(() => { // Gravity bunny.y += velocityY; velocityY += gravity; // Rotation while jumping if (isJumping) { bunny.rotation += 1; } if (bunny.y >= 370) { bunny.y = 370; velocityY = 0; isJumping = false; bunny.rotation = 0; } // Move obstacles for (let i = obstacles.length - 1; i >= 0; i--) { const obs = obstacles[i]; obs.x -= 5; // Collision detection const hit = bunny.x < obs.x + obs.width && bunny.x + bunny.width > obs.x && bunny.y < obs.y + obs.height && bunny.y + bunny.height > obs.y; if (hit) { alert("Game Over! Final Score: " + score); window.location.reload(); } // Remove off-screen if (obs.x + obs.width < 0) { app.stage.removeChild(obs); obstacles.splice(i, 1); score++; scoreText.text = `Score: ${score}`; } } }); })();
